{{- /* 惑星の一覧を、表示に必要な情報つきで返す（partialCached で呼ぶ） - data/planets.yaml の各惑星に、url・作品数（count）・表示するか（visible）・準備中か（pending）を足す - 作品数：works は planets に惑星IDを含む作品、fanart は /fanart/ 配下のページ - 本番：作品数が min_works 未満の惑星は非表示。確認用URL（本番以外）：すべて表示し、足りない惑星は pending - 二次創作エリアを非公開（config.toml の fanart_public = false）にしているときは、fanart を常に非表示 ※ 一覧ページを作るかどうかは content/_content.gotmpl で同じ規則で決めている */ -}} {{- $min := site.Params.min_works | default 1 -}} {{- $works := where (where site.RegularPages "Section" "works") "Params.planets" "!=" nil -}} {{- $list := slice -}} {{- range hugo.Data.planets -}} {{- $id := .id -}} {{- $count := 0 -}} {{- $url := printf "/%s/" $id -}} {{- $public := true -}} {{- if eq $id "fanart" -}} {{- $public = site.Params.fanart_public | default false -}} {{- with site.GetPage "/fanart" -}}{{- $count = len .RegularPages -}}{{- end -}} {{- else -}} {{- $count = len (where $works "Params.planets" "intersect" (slice $id)) -}} {{- end -}} {{- $enough := ge $count $min -}} {{- $visible := and $public (or $enough (not hugo.IsProduction)) -}} {{- $list = $list | append (merge . (dict "url" $url "count" $count "visible" $visible "pending" (not $enough))) -}} {{- end -}} {{- return $list -}}
